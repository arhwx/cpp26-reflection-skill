Program = { Let } Expr ;
Let = "let" identifier "=" Expr ";" ;
Expr = Term { AddOp } ;
Term = Factor { MulOp } ;
AddOp = ( "+" | "-" ) Term ;
Factor = [ "-" ] ( Num | Call | Var | Group ) ;
MulOp = ( "*" | "/" ) Factor ;
Num = integer ;
Call = identifier "(" Expr ")" ;
Var = identifier ;
Group = "(" Expr ")" ;
--
Value = ( "true" | "false" | integer | List ) ;
List = "[" [ Items ] "]" ;
Items = Value { More } ;
More = "," Value ;
